41/// ```c

43/// ```
44unsafe extern "C" fn strhash(arg: st_data_t) -> st_index_t {
45    let string = CStr::from_ptr(arg.as_const_c_char());
46    let mut hasher = FnvHasher::default();
47    hasher.write(string.to_bytes());
48    hasher.finish().into()
49}

61unsafe extern "C" fn strcasehash(arg: st_data_t) -> st_index_t {
62    let string = CStr::from_ptr(arg.as_const_c_char());
63    let mut hasher = FnvHasher::default();
64    for byte in string.to_bytes() {
65        hasher.write_u8(byte.to_ascii_lowercase());
66    }
67    hasher.finish().into()
68}

135/// # Header declaration
136///
137/// ```c
138/// PUREFUNC(int st_locale_insensitive_strcasecmp(const char *s1, const char *s2));
139/// ```
140#[no_mangle]
141unsafe extern "C" fn st_locale_insensitive_strcasecmp(s1: st_data_t, s2: st_data_t) -> libc::c_int {
142    let s1 = CStr::from_ptr(s1.as_const_c_char());
143    let s2 = CStr::from_ptr(s2.as_const_c_char());
144    match (s1.to_bytes().len(), s2.to_bytes().len()) {
145        (left, right) if left == right => {}
146        (left, right) if left > right => return 1,
147        _ => return -1,
148    }
149
150    for (&left, &right) in s1.to_bytes().iter().zip(s2.to_bytes().iter()) {
151        // there are guaranteed to be no interior NULs in this loop
152        let c1 = left.to_ascii_lowercase();
153        let c2 = right.to_ascii_lowercase();
154        match (c1, c2) {
155            (a, b) if a == b => {}
156            (a, b) if a > b => return 1,
157            _ => return -1,
158        }
159    }
160    0
161}

163/// # Header declaration
164///
165/// ```c
166/// PUREFUNC(int st_locale_insensitive_strncasecmp(const char *s1, const char *s2, size_t n));
167/// ```
168#[no_mangle]
169unsafe extern "C" fn st_locale_insensitive_strncasecmp(
170    s1: st_data_t,
171    s2: st_data_t,
172    n: libc::size_t,
173) -> libc::c_int {
174    let s1 = slice::from_raw_parts(s1.as_const_c_char().cast::<u8>(), n);
175    let s2 = slice::from_raw_parts(s2.as_const_c_char().cast::<u8>(), n);
176
177    for (&left, &right) in iter::zip(s1, s2) {
178        match (left, right) {
179            (b'\0', b'\0') => return 0,
180            (_, b'\0') => return 1,
181            (b'\0', _) => return -1,
182            (c1, c2) => match (c1.to_ascii_lowercase(), c2.to_ascii_lowercase()) {
183                (a, b) if a == b => {}
184                (a, b) if a > b => return 1,
185                _ => return -1,
186            },
187        }
188    }
189    0
190}
