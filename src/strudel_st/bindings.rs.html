23/// Equality comparator and hash function used to build a [`StHashMap`] hasher.
24///
25/// These functions are `unsafe extern "C" fn` and expected to be supplied via
26/// FFI.
27///
28/// # Safety
29///
30/// `st_hash_type` are expected to have `'static` lifetime. This assumption is
31/// exploited by [`StHashMap`] and [`StBuildHasher`].
32#[repr(C)]
33#[derive(Debug, Clone, Copy)]
34pub struct st_hash_type {
35    /// `st_compare_func`
36    ///
37    /// # Header declaration
38    ///
39    /// ```c
40    /// (*compare)(ANYARGS /*st_data_t, st_data_t*/); /* st_compare_func* */
41    /// ```
42    pub compare: st_compare_func,
43
44    /// `st_hash_func`
45    ///
46    /// # Header declaration
47    ///
48    /// ```c
49    /// st_index_t (*hash)(ANYARGS /*st_data_t*/);        /* st_hash_func* */
50    /// ```
51    pub hash: st_hash_func,
52}

54/// Return values from [`st_foreach_callback_func`] and
55/// [`st_update_callback_func`] callback function pointers.
56#[repr(C)]
57#[derive(Debug, Clone, Copy, Hash, PartialEq, Eq, PartialOrd, Ord)]
58#[allow(clippy::upper_case_acronyms)]
59pub enum st_retval {
60    /// Continue iteration.
61    ST_CONTINUE,
62
63    /// Stop iteration.
64    ST_STOP,
65
66    /// Delete current iteration `(key, value)` pair and continue iteration.
67    ST_DELETE,
68
69    /// Continue or stop iteration.
70    ///
71    /// This return value has slightly different behavior depending on API. See
72    /// [`st_foreach`] and [`st_foreach_check`].
73    ///
74    /// [`st_foreach`]: crate::ffi::st_foreach
75    /// [`st_foreach_check`]: crate::ffi::st_foreach_check
76    ST_CHECK,
77}
78
79impl PartialEq<i32> for st_retval {
80    fn eq(&self, other: &i32) -> bool {
81        *self as i32 == *other
82    }
83}
84
85impl PartialEq<st_retval> for i32 {
86    fn eq(&self, other: &st_retval) -> bool {
87        *self == *other as i32
88    }
89}
