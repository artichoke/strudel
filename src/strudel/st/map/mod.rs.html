109/// An insertion-ordered hash map implemented with [`HashMap`] and [`Vec`].
110///
111/// `StHashMap` is designed to implement the `st_hash` C API and be
112/// FFI-friendly.
113///
114/// `StHashMap` is built on top of a hashing algorithm selected to provide
115/// resistance against HashDoS attacks. See [`RandomState`].
116///
117/// `StHashMap` supports updating keys in place. See [`StHashMap::update`].
118///
119/// The optional `api` and `capi` modules in `strudel` build on top of
120/// `StHashMap` to implement a compatible C API to `st_hash`. This API includes
121/// support for iterating over a mutable map and inplace updates of
122/// `(key, value)` pairs. These features distinguish it from the [`HashMap`] in
123/// Rust `std`.
124///
125/// [`RandomState`]: std::collections::hash_map::RandomState
126#[derive(Default, Debug, Clone)]
127#[allow(clippy::module_name_repetitions)]
128pub struct StHashMap<K, V, S = RandomState> {
129    map: HashMap<Key<K>, Value<V>, S>,
130    ordered: Vec<InsertionEntry<K, V>>,
131}
132

366    /// An iterator for visiting all insertion counters in insertion order
367    /// starting from the given rank. The iterator element type is `usize`.
368    ///
369    /// This iterator may return insertion counters to dead elements.
370    ///
371    /// The yielded elements may be passed to [`get_nth`] to retrieve the
372    /// `(key, value)` pair in the nth insertion slot.
373    ///
374    /// This API can be used to build a mutable iterator over the map that can
375    /// safely be invalidated. This is safe because new inserts always have
376    /// higher insert rank. See `api::st_foreach`.
377    ///
378    /// # Examples
379    ///
380    /// ```
381    /// use strudel::StHashMap;
382    ///
383    /// let mut map = StHashMap::new();
384    /// map.insert("a", 1);
385    /// map.insert("b", 2);
386    /// map.insert("c", 3);
387    ///
388    /// map.remove(&"a");
389    /// map.insert("b", 100);
390    ///
391    /// let insert_ranks = map.insert_ranks_from(0).collect::<Vec<_>>();
392    /// assert_eq!(vec![0, 1, 2], insert_ranks);
393    ///
394    /// assert_eq!(None, map.get_nth(0));
395    /// assert_eq!(Some((&"b", &100)), map.get_nth(1));
396    /// assert_eq!(Some((&"c", &3)), map.get_nth(2));
397    /// assert_eq!(None, map.get_nth(4));
398    ///
399    /// assert_eq!(0, map.insert_ranks_from(100).count());
400    /// ```
401    ///
402    /// [`get_nth`]: StHashMap::get_nth
403    #[inline]
404    #[must_use]
405    pub fn insert_ranks_from(&self, rank: usize) -> InsertRanks {
406        InsertRanks(rank..self.ordered.len())
407    }

532    pub fn min_insert_rank(&self) -> usize {
533        for (idx, pair) in self.ordered.iter().enumerate() {
534            if let InsertionEntry::Alive(_, _) = pair {
535                return idx;
536            }
537        }
538        0
539    }

645    /// Return an estimate of the byte size of memory allocted for this map.
646    ///
647    /// # Examples
648    ///
649    /// ```
650    /// use strudel::StHashMap;
651    /// let empty: StHashMap<i32, i32> = StHashMap::with_capacity(0);
652    /// let map: StHashMap<i32, i32> = StHashMap::with_capacity(100);
653    /// assert!(map.estimated_memsize() > empty.estimated_memsize());
654    /// ```
655    #[inline]
656    #[must_use]
657    pub fn estimated_memsize(&self) -> usize {
658        let stack_size = size_of::<Self>();
659        let hashmap_size = (size_of::<Key<K>>() + size_of::<V>()) * self.map.capacity();
660        let vec_size = size_of::<InsertionEntry<K, V>>() * self.ordered.len();
661
662        stack_size + hashmap_size + vec_size
663    }

723    /// map.insert(1, "a");
724    /// assert_eq!(map.contains_key(&1), true);
725    /// assert_eq!(map.contains_key(&2), false);
726    /// ```
727    #[inline]
728    #[must_use]
729    pub fn contains_key(&self, key: &K) -> bool {
730        self.map.contains_key(key)
731    }

733    /// Returns a reference to the value corresponding to the key.
734    ///
735    /// # Examples
736    ///
737    /// ```
738    /// use strudel::StHashMap;
739    ///
740    /// let mut map = StHashMap::new();
741    /// map.insert(1, "a");
742    /// assert_eq!(map.get(&1), Some(&"a"));
743    /// assert_eq!(map.get(&2), None);
744    /// ```
745    #[inline]
746    #[must_use]
747    pub fn get(&self, key: &K) -> Option<&V> {
748        if self.ordered.len() < LINEAR_SCAN_THRESHOLD {
749            self.ordered.iter().find_map(|entry| match entry {
750                InsertionEntry::Alive(entry_key, entry_value) if entry_key == key => {
751                    Some(entry_value)
752                }
753                _ => None,
754            })
755        } else {
756            self.map.get(key).map(Value::inner)
757        }
758    }

760    /// Returns the key-value pair corresponding to the supplied key.
761    ///
762    /// # Examples
763    ///
764    /// ```
765    /// use strudel::StHashMap;
766    ///
767    /// let mut map = StHashMap::new();
768    /// map.insert(1, "a");
769    /// assert_eq!(map.get_key_value(&1), Some((&1, &"a")));
770    /// assert_eq!(map.get_key_value(&2), None);
771    /// ```
772    #[inline]
773    #[must_use]
774    pub fn get_key_value(&self, key: &K) -> Option<(&K, &V)> {
775        let (key, value) = self.map.get_key_value(key)?;
776        Some((key.inner(), value.inner()))
777    }

780impl<K, V, S> StHashMap<K, V, S>
781where
782    K: Eq + Hash + Clone,
783    V: PartialEq + Clone,
784    S: BuildHasher,
785{
786    /// Inserts a key-value pair into the map.
787    ///
788    /// If the map did not have this key present, [`None`] is returned.
789    ///
790    /// If the map did have this key present, the value is updated, and the old
791    /// value is returned. The key is not updated, though. To update the key
792    /// in-place, use [`StHashMap::update`].
793    #[inline]
794    #[must_use]
795    #[allow(clippy::missing_panics_doc)]
796    pub fn insert(&mut self, key: K, value: V) -> Option<V> {
797        let insert_rank = self.ordered.len();
798
799        let key = Key {
800            inner: key,
801            insert_rank,
802        };
803
804        match self.map.entry(key) {
805            HashEntry::Occupied(mut base) => {
806                let insert_rank = base.key().insert_rank();
807                // Maintain insert rank with new key-value pair.
808                if let Some(InsertionEntry::Alive(_, stored_value)) =
809                    self.ordered.get_mut(insert_rank)
810                {
811                    if *stored_value != value {
812                        *stored_value = value.clone();
813                    }
814                } else {
815                    panic!("already inserted pair not alive in ordered storage");
816                }
817                Some(base.insert(value.into()).into_inner())
818            }
819            HashEntry::Vacant(base) => {
820                self.ordered.push(InsertionEntry::Alive(
821                    base.key().inner().clone(),
822                    value.clone(),
823                ));
824                base.insert(value.into());
825                None
826            }
827        }
828    }
829
830    /// Inserts a key-value pair into the map and update the key in place if an
831    /// entry is already present.
832    ///
833    /// This function maintains the insertion rank of the key-value pair.
834    ///
835    /// If you do not wish to update the key in-place, use
836    /// [`StHashMap::insert`].
837    #[allow(clippy::missing_panics_doc)]
838    #[inline]
839    pub fn update(&mut self, key: K, value: V) {
840        if let Some((entry_key, _)) = self.map.remove_entry(&key) {
841            let insert_rank = entry_key.insert_rank();
842            // Maintain insert rank with new key-value pair.
843            if let Some(InsertionEntry::Alive(stored_key, stored_value)) =
844                self.ordered.get_mut(insert_rank)
845            {
846                if *stored_key != key {
847                    *stored_key = key.clone();
848                }
849                if *stored_value != value {
850                    *stored_value = value.clone();
851                }
852            } else {
853                panic!("already inserted pair not alive in ordered storage");
854            }
855            let key = Key {
856                inner: key,
857                insert_rank,
858            };
859            self.map.insert(key, value.into());
860        } else {
861            let _ = self.insert(key, value);
862        }
863    }
864
865    /// Removes a key from the map, returning the stored key if the key was
866    /// previously in the map.
867    ///
868    /// # Examples
869    ///
870    /// ```
871    /// use strudel::StHashMap;
872    ///
873    /// let mut map = StHashMap::new();
874    /// map.insert(1, "a");
875    /// assert_eq!(map.remove(&1), Some("a"));
876    /// assert_eq!(map.remove(&1), None);
877    /// ```
878    #[inline]
879    #[must_use]
880    pub fn remove(&mut self, key: &K) -> Option<V> {
881        self.remove_entry(key).map(|(_, value)| value)
882    }
883
884    /// Removes a key from the map, returning the stored key and value if the
885    /// key was previously in the map.
886    ///
887    /// # Examples
888    ///
889    /// ```
890    /// use strudel::StHashMap;
891    ///
892    /// let mut map = StHashMap::new();
893    /// map.insert(1, "a");
894    /// assert_eq!(map.remove_entry(&1), Some((1, "a")));
895    /// assert_eq!(map.remove(&1), None);
896    /// ```
897    #[inline]
898    #[must_use]
899    pub fn remove_entry(&mut self, key: &K) -> Option<(K, V)> {
900        let (key, value) = self.map.remove_entry(key)?;
901        self.ordered[key.insert_rank()] = InsertionEntry::Dead;
902        Some((key.into_inner(), value.into_inner()))
903    }
